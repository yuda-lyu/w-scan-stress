// 資源記錄:各服務(依埠找監聽之行程)之累計處理器時間與記憶體, 以及整機之處理器使用率與記憶體(含 k6 本身)
// 僅支援 Windows(監聽者之查詢經 wsemi 之 execProcessPortPidsSync: netstat 與 Get-NetTCPConnection; 行程資源以 tasklist)
import os from 'node:os'
import { execSync } from 'node:child_process'
import execProcessPortPidsSync from 'wsemi/src/execProcessPortPidsSync.mjs'


// tasklist 之輸出雖小, 仍明設 maxBuffer; stderr 不外洩至主控台
// (監聽者之查詢不在此執行: 施壓時 netstat 之輸出可達數 MB, 其 maxBuffer 由 wsemi 之 execProcessPortPidsSync 設定)
const EXEC_OPT = { encoding: 'utf8', maxBuffer: 256 * 1024 * 1024, stdio: ['ignore', 'pipe', 'ignore'] }


/**
 * 查詢監聽指定埠之行程PID(經wsemi之execProcessPortPidsSync: Windows先netstat，查無或失敗再以Get-NetTCPConnection確認；判定不依netstat之狀態欄，因本機OS之TCP狀態值有錯置、netstat亦會漏列新起之監聽)
 *
 * 空陣列只代表已確認無人監聽；查詢失敗(指令無法執行、逾時、輸出無法解析)與埠號不合法一律拋錯，不以空陣列代替
 *
 * @param {Number} port 輸入埠號，須為1~65535之整數
 * @param {Object} [opt={}] 輸入設定物件，供測試注入，僅轉傳exec(spawnSync形之執行函數)、platform給execProcessPortPidsSync(格式見其說明)
 * @returns {Array} 回傳PID數字陣列(去重、由小到大)，空陣列代表無人監聽
 * @throws {Error} 埠號不合法或查詢失敗時拋錯，訊息為查詢失敗之原因
 */
export function pidOfPort(port, opt = {}) {
    if (!Number.isInteger(port) || port < 1 || port > 65535) {
        throw new Error(`port 須為 1~65535 之整數:${String(port)}`)
    }
    //失敗之原因僅由 onTrace 之 settle 事件取得(同步版回傳 null 而不拋錯)
    let msg = ''
    const pids = execProcessPortPidsSync(port, {
        exec: opt.exec,
        platform: opt.platform,
        onTrace: (e) => {
            if (e.type === 'settle' && !e.ok) msg = e.msg
        },
    })
    if (pids === null) throw new Error(msg || `port[${port}] listener query failed`)
    return pids
}


/**
 * 以tasklist取行程之記憶體(工作集，MB)與累計處理器時間(秒)，欄位依位置取(不依語系之欄名)
 *
 * @param {String} pid 輸入PID字串
 * @returns {Object|null} 回傳{pid,memMb,cpuSec}，取不到則回傳null
 */
export function procStat(pid) {
    if (!/^\d+$/.test(String(pid))) return null
    try {
        const out = execSync(`tasklist /V /FO CSV /NH /FI "PID eq ${pid}"`, EXEC_OPT)
        const cols = [...out.matchAll(/"([^"]*)"/g)].map((m) => m[1])
        if (cols.length < 8) return null
        const memMb = Number(cols[4].replace(/[^\d]/g, '')) / 1024
        const [h, mi, s] = cols[7].split(':').map(Number)
        return { pid, memMb: Math.round(memMb), cpuSec: h * 3600 + mi * 60 + s }
    }
    catch {
        return null
    }
}


/**
 * 依行程之埠取各行程之資源
 *
 * 同一埠有多個行程監聽時取PID最小者(pidOfPort之結果已排序，前後兩次取樣取同一個)；本機OS上新起之監聽或無人之埠，每埠查詢約1秒(netstat漏列而改以Get-NetTCPConnection確認)
 *
 * @param {Object} kpSvcPort 輸入行程鍵名對埠號之物件，例如{api:8000,auth:8001}
 * @param {Object} [opt={}] 輸入設定物件，供測試注入，轉傳給pidOfPort
 * @returns {Object} 回傳行程鍵名對{pid,memMb,cpuSec}之物件；無人監聽該埠或tasklist取不到者為null；查詢監聽者失敗者為{error}，error為失敗原因字串(不與無人監聽混淆)
 */
export function snapProcs(kpSvcPort, opt = {}) {
    const r = {}
    for (const k of Object.keys(kpSvcPort)) {
        try {
            const pids = pidOfPort(kpSvcPort[k], opt)
            r[k] = pids.length > 0 ? procStat(String(pids[0])) : null
        }
        catch (err) {
            r[k] = { error: err.message }
        }
    }
    return r
}


/**
 * 列出一次資源取樣中查詢監聽者失敗之行程
 *
 * @param {Object} snap 輸入snapProcs之回傳物件(或sys.json之procBefore、procAfter)
 * @returns {Array} 回傳字串陣列，每項為「行程鍵名: 失敗原因」，無則為空陣列
 */
export function listSnapErrors(snap) {
    if (!snap || typeof snap !== 'object') return []
    return Object.keys(snap).filter((k) => snap[k] && typeof snap[k].error === 'string').map((k) => `${k}: ${snap[k].error}`)
}


/**
 * 開始每秒取樣整機之處理器使用率(%)與記憶體使用量(GB)，回傳停止函數，呼叫後回傳取樣陣列[{t,cpu,memGb}]
 *
 * @returns {Function} 回傳停止取樣之函數
 */
export function startSampler() {
    const cpuTimes = () => os.cpus().reduce((a, c) => {
        const t = c.times
        a.idle += t.idle
        a.total += t.user + t.nice + t.sys + t.idle + t.irq
        return a
    }, { idle: 0, total: 0 })
    const samples = []
    let prev = cpuTimes()
    const tm = setInterval(() => {
        const cur = cpuTimes()
        const dt = cur.total - prev.total
        const cpu = dt > 0 ? (1 - (cur.idle - prev.idle) / dt) * 100 : 0
        prev = cur
        samples.push({ t: Date.now(), cpu: Math.round(cpu), memGb: Math.round((os.totalmem() - os.freemem()) / 1024 / 1024 / 102.4) / 10 })
    }, 1000)
    return () => {
        clearInterval(tm)
        return samples
    }
}
