// 情境(scenario)與 session:使用方傳入之請求動線, 格式見 k6Scenario.js 之說明; 本模組檢查其格式、摘要其內容,
// 並處理 session: 遮蔽日誌與 meta 中 session 之值、釋放 session(釋放失敗不遮蓋呼叫端之錯誤, 也不遺失 session)
import fs from 'node:fs'
import { RESERVED_METRICS } from './config.mjs'


// k6 之 b64decode(標準編碼)接受之字串: 4 字元一組, 末組可為 2 字元加 == 或 3 字元加 =, 空字串亦可
const B64_RE = /^(?:[A-Za-z0-9+/]{4})*(?:[A-Za-z0-9+/]{2}==|[A-Za-z0-9+/]{3}=)?$/
const isObj = (v) => !!v && typeof v === 'object' && !Array.isArray(v)
const isNum = (v, min) => typeof v === 'number' && Number.isFinite(v) && v >= min
const isPosInt = (v) => Number.isInteger(v) && v > 0

// 陣列之各索引皆須檢核: Array.from 將稀疏陣列之空位轉為 undefined, 避免 forEach、every 跳過空位而於序列化後變成 null
const dense = (a) => Array.from(a)

// equals 須可序列化且可用 === 比較: 字串、有限數字、布林或 null(undefined 於序列化時消失, 缺值會變成「缺少的欄位等於缺值」而通過)
const isEq = (v) => v === null || typeof v === 'string' || typeof v === 'boolean' || (typeof v === 'number' && Number.isFinite(v))


/**
 * 檢查情境之格式，有誤即拋錯(於施壓前檢查，不做白工)
 *
 * 凡k6Scenario.js讀取之欄位皆須檢查，否則錯誤只會在施壓開始後於k6內以執行期例外出現
 *
 * @param {Object} sc 輸入情境物件{bodies,bodiesPerVu,phases}
 * @returns {Object} 回傳原情境物件
 */
export function validateScenario(sc) {
    if (!sc || typeof sc !== 'object') throw new Error('情境須為物件')
    if (!Array.isArray(sc.phases) || sc.phases.length === 0) throw new Error('情境須有 phases(非空陣列)')
    if (sc.bodies !== undefined && !isObj(sc.bodies)) throw new Error('bodies 須為物件')
    if (sc.bodiesPerVu !== undefined && !isObj(sc.bodiesPerVu)) throw new Error('bodiesPerVu 須為物件')
    const bodies = sc.bodies || {}
    const bodiesPerVu = sc.bodiesPerVu || {}
    for (const k of Object.keys(bodies)) {
        if (typeof bodies[k] !== 'string' || !B64_RE.test(bodies[k])) throw new Error(`bodies.${k} 須為 base64 字串(標準編碼, 含補齊之 =)`)
    }
    for (const k of Object.keys(bodiesPerVu)) {
        if (!Array.isArray(bodiesPerVu[k]) || bodiesPerVu[k].length === 0) throw new Error(`bodiesPerVu.${k} 須為非空陣列`)
        dense(bodiesPerVu[k]).forEach((b, i) => {
            if (typeof b !== 'string' || !B64_RE.test(b)) throw new Error(`bodiesPerVu.${k}[${i}] 須為 base64 字串(標準編碼, 含補齊之 =)`)
        })
    }
    const ckReq = (r, where) => {
        if (!r || typeof r !== 'object') throw new Error(`${where}: 請求須為物件`)
        if (typeof r.name !== 'string' || !/^[\w.:-]+$/.test(r.name) || r.name === '__proto__') throw new Error(`${where}: 請求之 name 須為英數字、底線、點、冒號或連字號(不得為 __proto__):${r.name}`)
        if (!r.url) throw new Error(`${where}: 請求須有 url`)
        if (typeof r.url !== 'string') throw new Error(`${where}: url 須為字串`)
        if (r.method !== undefined && !(typeof r.method === 'string' && /^[A-Za-z][\w-]*$/.test(r.method))) throw new Error(`${where}: method 須為英文字母開頭、僅含英數字、底線與連字號之字串:${r.method}`)
        if (r.headers !== undefined && !isObj(r.headers)) throw new Error(`${where}: headers 須為物件`)
        if (r.body !== undefined && r.body !== null) {
            if (!isObj(r.body)) throw new Error(`${where}: body 須為物件({ ref }、{ refPerVu } 或 { text })`)
            if (r.body.ref !== undefined && !Object.hasOwn(bodies, r.body.ref)) throw new Error(`${where}: body.ref 不在 bodies:${r.body.ref}`)
            if (r.body.refPerVu !== undefined && !Object.hasOwn(bodiesPerVu, r.body.refPerVu)) throw new Error(`${where}: body.refPerVu 不在 bodiesPerVu:${r.body.refPerVu}`)
            if (r.body.text !== undefined && typeof r.body.text !== 'string') throw new Error(`${where}: body.text 須為字串`)
        }
        const c = r.check
        if (c !== undefined && c !== null) {
            if (!isObj(c)) throw new Error(`${where}: check 須為物件`)
            if (c.status !== undefined && !(Array.isArray(c.status) && c.status.length > 0 && dense(c.status).every(Number.isInteger))) throw new Error(`${where}: check.status 須為非空之整數陣列`)
            if (c.header !== undefined) {
                if (!(isObj(c.header) && typeof c.header.name === 'string' && c.header.name)) throw new Error(`${where}: check.header 須為有 name 之物件`)
                if (typeof c.header.equals !== 'string') throw new Error(`${where}: check.header.equals 須為字串`)
            }
            if (c.json !== undefined) {
                if (!Array.isArray(c.json)) throw new Error(`${where}: check.json 須為陣列`)
                dense(c.json).forEach((q, i) => {
                    if (!isObj(q) || typeof q.path !== 'string' || !q.path) throw new Error(`${where}: check.json[${i}] 須有 path`)
                    if (!isEq(q.equals)) throw new Error(`${where}: check.json[${i}].equals 須為字串、有限數字、布林或 null(缺少時缺欄位會被視為相等)`)
                })
            }
        }
    }
    dense(sc.phases).forEach((ph, i) => {
        const where = `phases[${i}]`
        if (!isObj(ph)) throw new Error(`${where} 須為物件`)
        if (ph.name !== undefined && typeof ph.name !== 'string') throw new Error(`${where}: name 須為字串`)
        if (ph.kind === 'requests') {
            if (!Array.isArray(ph.steps) || ph.steps.length === 0) throw new Error(`${where}: requests 須有 steps(非空陣列)`)
            if (ph.timer !== undefined) {
                if (typeof ph.timer !== 'string' || !/^[a-zA-Z_]\w*$/.test(ph.timer)) throw new Error(`${where}: timer 須為英數字與底線且不以數字開頭:${ph.timer}`)
                if (RESERVED_METRICS.includes(ph.timer) || ph.timer === '__proto__') throw new Error(`${where}: timer 不得與 k6 內建指標或 app_errors 同名(亦不得為 __proto__):${ph.timer}`)
                if (/_ok$/.test(ph.timer)) throw new Error(`${where}: timer 不得以 _ok 結尾:${ph.timer}`)
            }
            dense(ph.steps).forEach((st, j) => {
                if (!Array.isArray(st)) throw new Error(`${where}.steps[${j}] 須為請求之陣列`)
                dense(st).forEach((r, k) => ckReq(r, `${where}.steps[${j}][${k}]`))
            })
        }
        else if (ph.kind === 'stay') {
            if (!isPosInt(ph.sec)) throw new Error(`${where}: stay 須有 sec(正整數, 以秒計, k6 腳本逐秒執行)`)
            if (!Array.isArray(ph.ticks)) throw new Error(`${where}: stay 須有 ticks(陣列)`)
            dense(ph.ticks).forEach((tk, j) => {
                if (!isObj(tk)) throw new Error(`${where}.ticks[${j}] 須為物件`)
                if (!isPosInt(tk.every)) throw new Error(`${where}.ticks[${j}]: every 須為正整數(以秒計)`)
                if (!Array.isArray(tk.reqs)) throw new Error(`${where}.ticks[${j}]: reqs 須為請求之陣列`)
                dense(tk.reqs).forEach((r, k) => ckReq(r, `${where}.ticks[${j}].reqs[${k}]`))
            })
        }
        else if (ph.kind === 'sleep') {
            if (!isNum(ph.sec, 0)) throw new Error(`${where}: sleep 須有 sec(非負數)`)
        }
        else {
            throw new Error(`${where}: kind 須為 requests、stay 或 sleep:${ph.kind}`)
        }
    })
    return sc
}


/**
 * 摘要情境：各階段之種類、名稱、計時與請求數，以及全部分組名稱與計時名稱
 *
 * @param {Object} sc 輸入情境物件
 * @returns {Object} 回傳{phases:[{kind,name,timer,nReqs}],names,timers}
 */
export function summarizeScenario(sc) {
    const names = new Set()
    const timers = []
    const phases = sc.phases.map((ph) => {
        let reqs = []
        if (ph.kind === 'requests') reqs = ph.steps.flat()
        if (ph.kind === 'stay') reqs = ph.ticks.flatMap((tk) => tk.reqs)
        reqs.forEach((r) => names.add(r.name))
        if (ph.timer) timers.push(ph.timer)
        return { kind: ph.kind, name: ph.name || '', timer: ph.timer || '', nReqs: reqs.length }
    })
    return { phases, names: [...names], timers }
}


/**
 * 由各session取出須遮蔽之值(字串且長度8以上者)與其變數名
 *
 * @param {Array} sessions 輸入session物件陣列
 * @returns {Array} 回傳[{key,value}]，依值之長度由長至短
 */
export function collectSecrets(sessions) {
    const out = []
    const seen = new Set()
    for (const ss of sessions || []) {
        for (const [k, v] of Object.entries(ss || {})) {
            if (typeof v === 'string' && v.length >= 8 && !seen.has(v)) {
                seen.add(v)
                out.push({ key: k, value: v })
            }
        }
    }
    return out.sort((a, b) => b.value.length - a.value.length)
}


/**
 * 遮蔽文字中之session值為<變數名>
 *
 * @param {String} text 輸入文字
 * @param {Array} secrets 輸入collectSecrets之結果
 * @returns {String} 回傳遮蔽後之文字
 */
export function maskSecrets(text, secrets) {
    let t = String(text)
    for (const { key, value } of secrets) {
        if (t.includes(value)) t = t.split(value).join(`<${key}>`)
    }
    return t
}


/**
 * 遞迴遮蔽物件(陣列)中各字串(含鍵名)之session值，供序列化前使用
 *
 * 先遮蔽再序列化，使佔位符與原值中需跳脫之字元(引號、反斜線)由JSON.stringify統一處理；對已序列化之JSON字串做字串替換，則會漏掉已跳脫之值，也可能寫出非法JSON
 *
 * @param {*} v 輸入任意值
 * @param {Array} secrets 輸入collectSecrets之結果
 * @returns {*} 回傳遮蔽後之複本
 */
export function maskDeep(v, secrets) {
    if (typeof v === 'string') return maskSecrets(v, secrets)
    if (Array.isArray(v)) return v.map((x) => maskDeep(x, secrets))
    if (v && typeof v === 'object') return Object.fromEntries(Object.entries(v).map(([k, x]) => [maskSecrets(k, secrets), maskDeep(x, secrets)]))
    return v
}


/**
 * 遮蔽檔案內容中之session值(原地改寫)，檔案不存在則略過
 *
 * @param {String} fp 輸入檔案路徑字串
 * @param {Array} secrets 輸入collectSecrets之結果
 * @returns {Boolean} 回傳是否有改寫布林值
 */
export function maskFile(fp, secrets) {
    if (!fs.existsSync(fp)) return false
    const t = fs.readFileSync(fp, 'utf8')
    const m = maskSecrets(t, secrets)
    if (m === t) return false
    fs.writeFileSync(fp, m, 'utf8')
    return true
}


/**
 * 釋放session，回傳未能釋放者
 *
 * releaseFn須回傳未能釋放者之陣列(全部釋放則為空陣列)；拋錯或回傳非陣列(含未回傳值)時，視為全部未能釋放：不遮蓋呼叫端之原始錯誤，也不遺失session，使其留於pending檔供另行處理
 *
 * @param {Function} releaseFn 輸入釋放session之函數，即opt.releaseSessions
 * @param {Array} sessions 輸入session物件陣列
 * @param {Array} secrets 輸入collectSecrets之結果，用於遮蔽訊息中session之值
 * @param {Function} log 輸入訊息輸出函數
 * @returns {Promise} 回傳Promise，resolve回傳未能釋放之session陣列，不會reject
 */
export async function releaseAll(releaseFn, sessions, secrets, log) {
    try {
        const r = await releaseFn(sessions)
        if (Array.isArray(r)) return r
        log('opt.releaseSessions 須回傳未能釋放者之陣列(全部釋放則為空陣列), 視為全部未能釋放')
    }
    catch (err) {
        log(`opt.releaseSessions 拋錯, 視為全部未能釋放: ${maskSecrets(String((err && err.message) || err).split('\n')[0], secrets)}`)
    }
    return sessions
}
